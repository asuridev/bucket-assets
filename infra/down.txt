#!/bin/sh
# Baja el stack que haya generado up.sh.
#
#   ./down.sh        para los contenedores. Los datos SOBREVIVEN.
#   ./down.sh -v     ademas borra los volumenes: se pierde todo lo que haya dentro.
set -e

cd "$(dirname "$0")"

COMPOSE_FILE=generated/docker-compose.yaml

die() { echo "ERROR: $*" >&2; exit 1; }

if [ ! -f "$COMPOSE_FILE" ]; then
  echo "No hay nada que bajar: no existe infra/$COMPOSE_FILE."
  echo "Ese fichero lo genera ./up.sh; si nunca lo ejecutaste, no hay stack levantado."
  exit 0
fi

DOWN_ARGS=""
case "$1" in
  -v|--volumes)
    # Confirmacion explicita: `down -v` es irreversible y se teclea por inercia detras de
    # un `down` normal.
    echo "Esto BORRA los volumenes del stack y todos los datos que contengan:"
    grep -A 10 '^volumes:' "$COMPOSE_FILE" | grep -E '^  [a-z-]+:' | sed 's/^/   - /' || true
    printf 'Escribe "si" para continuar: '
    read -r ANSWER || ANSWER=""
    [ "$ANSWER" = "si" ] || { echo "Cancelado."; exit 1; }
    DOWN_ARGS="-v"
    ;;
  "") ;;
  *) die "uso: $0 [-v]" ;;
esac

if command -v docker >/dev/null 2>&1 && docker compose version >/dev/null 2>&1; then
  COMPOSE="docker compose"
elif command -v podman-compose >/dev/null 2>&1; then
  COMPOSE="podman-compose"
else
  die "no encuentro ni 'docker compose' ni 'podman-compose'"
fi

echo "> $COMPOSE -p infra -f $COMPOSE_FILE down $DOWN_ARGS"
$COMPOSE -p infra -f "$COMPOSE_FILE" down $DOWN_ARGS

echo
echo "Stack parado. El compose generado sigue en infra/$COMPOSE_FILE por si quieres revisarlo;"
echo "./up.sh lo reescribe en la siguiente ejecucion."
