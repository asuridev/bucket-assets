#!/bin/sh
# Para el stack. Simetrico de up.sh.
#
#   ./deploy-mongo/down.sh        # para los contenedores. Los datos de Mongo SOBREVIVEN.
#   ./deploy-mongo/down.sh -v     # ademas BORRA el volumen: se pierde todo lo que haya en Mongo.
set -e

cd "$(dirname "$0")"

ENV_FILE=.env.devx
COMPOSE_ARGS="-f docker-compose.yaml"
[ -f "$ENV_FILE" ] && COMPOSE_ARGS="$COMPOSE_ARGS --env-file $ENV_FILE"

DOWN_ARGS=""
case "$1" in
  -v|--volumes)
    # Confirmacion explicita: `down -v` es irreversible y se teclea por inercia detras de un
    # `down` normal.
    printf 'Esto BORRA el volumen mongo-data y todos los datos de Mongo. Escribe "si" para continuar: '
    read -r ANSWER
    [ "$ANSWER" = "si" ] || { echo "Cancelado."; exit 1; }
    DOWN_ARGS="-v"
    ;;
  "") ;;
  *)
    echo "Uso: $0 [-v]" >&2
    exit 1
    ;;
esac

# docker compose donde lo haya; si no, podman-compose (Windows).
if command -v docker >/dev/null 2>&1 && docker compose version >/dev/null 2>&1; then
  COMPOSE="docker compose"
elif command -v podman-compose >/dev/null 2>&1; then
  COMPOSE="podman-compose"
else
  echo "ERROR: no encuentro ni 'docker compose' ni 'podman-compose'" >&2
  exit 1
fi

echo "> $COMPOSE $COMPOSE_ARGS down $DOWN_ARGS"
# shellcheck disable=SC2086
$COMPOSE $COMPOSE_ARGS down $DOWN_ARGS
