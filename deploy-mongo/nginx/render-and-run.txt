#!/bin/sh
# Genera el server block y cede el control al arranque normal de Bitnami.
#
# Existe como fichero aparte, y no como `command:` en el compose, a proposito: si el script
# viviera en el YAML habria que escribir el dolar de la variable como "$$" para que compose no
# lo expandiera, y hay implementaciones de compose (la de DevX) que no respetan esa convencion
# -> envsubst recibe basura y el contenedor muere nada mas arrancar. Aqui dentro el "$" es
# literal para todo el mundo.
#
# La imagen de Bitnami NO tiene el mecanismo de /etc/nginx/templates de la imagen oficial: hay
# que generar la config a mano y dejarla en server_blocks/.
set -e

TEMPLATE=/tmp/default.conf.template
TARGET=/opt/bitnami/nginx/conf/server_blocks/mongo-ui.conf

if [ ! -f "$TEMPLATE" ]; then
  echo "ERROR: no encuentro $TEMPLATE. Revisa la ruta del volumen en el compose:" >&2
  echo "       si el fichero no existe, Docker monta un DIRECTORIO vacio en su lugar." >&2
  exit 1
fi

# El argumento limita que variables se sustituyen. Sin el, envsubst se comeria $host, $scheme
# y el $1 del proxy_redirect, dejandolos vacios.
envsubst '$MONGO_UI_BASE_PATH' < "$TEMPLATE" > "$TARGET"

echo "server block generado en $TARGET (MONGO_UI_BASE_PATH='${MONGO_UI_BASE_PATH}')"

exec /opt/bitnami/scripts/nginx/entrypoint.sh /opt/bitnami/scripts/nginx/run.sh
